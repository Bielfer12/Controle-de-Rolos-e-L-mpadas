OU

Ao clicar no botao de incluir, aparece um popup com um listbox, aonde voce tem que botar qual cadastro que voce quer e ter que apertar em ok, ao fazer isso te leva direto para a inclusão daquele tal item. 

ex:

#Include "protheus.ch"

User Function CadastroDinamico()

    // Criando o formulário
    Local oView := FWFormView():New()
    Local cTipoProduto := "" // Variável para capturar o tipo de produto

    // Adicionando campos básicos
    oView:AddField("Código", "CODIGO", .T.)
    oView:AddField("Descrição", "DESC", .T.)
    oView:AddField("Data", "DATA", .T.)

    // ListBox para seleção do tipo de produto
    oView:AddField("Tipo", "TIPO", .T., { "Rolo", "Lâmpada", "Impressora", "Estação" }, { |cTipo| FWTrocaCampos(oView, cTipo) } )

    // Exibindo o formulário
    If oView:Show()
        MsgInfo("Cadastro salvo com sucesso!")
    Else
        MsgInfo("Cadastro cancelado.")
    EndIf

Return

// Função para manipular campos dinamicamente
Static Function FWTrocaCampos(oView, cTipo)
    // Limpando campos adicionais
    oView:DelField("CampoExtra1")
    oView:DelField("CampoExtra2")

    // Configurando novos campos baseados no tipo
    Do Case
        Case cTipo == "Rolo" .Or. cTipo == "Lâmpada"
            oView:AddField("CampoExtra1", "ESPECIFICAÇÃO", .T.)
            oView:AddField("CampoExtra2", "QUANTIDADE", .T.)

        Case cTipo == "Impressora" .Or. cTipo == "Estação"
            // Não adiciona campos extras
    EndCase

Return
